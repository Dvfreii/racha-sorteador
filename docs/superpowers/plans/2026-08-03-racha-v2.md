**Goal:** Refactor the society football team picker from SSR form-based to a REST API + vanilla JS SPA with unified player model, position array, restrictions, and WhatsApp export.

**Architecture:** Flask REST API with SQLAlchemy ORM (SQLite), vanilla JS SPA consuming JSON endpoints. Goleiro merged into Jogador via `is_goleiro` flag. **Tech Stack:** Flask 3.x, SQLAlchemy 2.x, SQLite, Python 3.11+, Vanilla JS (ES modules), Bootstrap 5.3 (CDN), pytest

- Soft delete only (ativo=False), no hard delete
- Legacy routes preserved in app.py for backward compat

```python
"""Migrate DB from v1 (Jogador + Goleiro separate) to v2 (Jogador unified + Posicao N:N)."""
from app import app
from backend.extensions import db
from backend.models.entities import Jogador, Goleiro, Posicao, SorteioJogador

POSICOES = [
    "Goleiro",
    "Zagueiro / Fixo",
    "Lateral",
    "Meio-Campo",
    "Alas",
    "Atacante / Pivo",
]


def migrate():
    with app.app_context():
        db.create_all()

        for nome in POSICOES:
            if not Posicao.query.filter_by(nome=nome).first():
                db.session.add(Posicao(nome=nome))
        db.session.commit()

        posicao_map = {p.nome.lower().replace(" / ", "/").replace("-", ""): p for p in Posicao.query.all()}

        # Migrate goleiros -> Jogador
        goleiros = Goleiro.query.filter_by(ativo=True).all()
        for g in goleiros:
            nome_normalizado = g.nome.strip()
            existing = Jogador.query.filter_by(nome=nome_normalizado, ativo=True).first()
            if existing:
                existing.is_goleiro = True
            else:
                j = Jogador(nome=nome_normalizado, nota=3.0, is_goleiro=True)
                db.session.add(j)
        db.session.commit()

        # Migrate old posicao string -> N:N
        for j in Jogador.query.filter_by(ativo=True).all():
            if not j.posicoes:
                old = getattr(j, "posicao", "")
                if old:
                    key = old.strip().lower().replace(" / ", "/").replace("-", "")
                    pos = posicao_map.get(key)
                    if pos:
                        j.posicoes.append(pos)
        db.session.commit()

        # Migrate SorteioGoleiro -> SorteioJogador.is_goleiro_no_time
        from sqlalchemy import text
        try:
            db.session.execute(text(
                "INSERT INTO sorteio_jogador (sorteio_id, jogador_id, time, is_goleiro_no_time) "
                "SELECT sg.sorteio_id, sg.goleiro_id, 'Goleiro', 1 FROM sorteio_goleiro sg "
                "WHERE NOT EXISTS (SELECT 1 FROM sorteio_jogador sj WHERE sj.sorteio_id = sg.sorteio_id AND sj.jogador_id = sg.goleiro_id)"
            ))
        except Exception:
            pass
        db.session.commit()

        print("Migration complete.")
        print(f"  Jogadores: {Jogador.query.filter_by(ativo=True).count()}")
        print(f"  Posicoes: {Posicao.query.count()}")
        print(f"  Sorteios: {db.session.execute(text('SELECT COUNT(*) FROM sorteio')).scalar()}")


if __name__ == "__main__":
    migrate()
```

```python
"""Motor de sorteio equilibrado do racha - v2."""
from collections import Counter
import random


def tamanhos_dos_times(total, quantidade=3, tamanhos=None):
    if quantidade < 2:
        raise ValueError("Escolha pelo menos 2 times.")
    if tamanhos:
        base = tamanhos[0]
        if base * (quantidade - 1) >= total:
            raise ValueError(
                f"Jogadores insuficientes: para {quantidade} times de {base}, "
                f"e preciso ter mais de {base * (quantidade - 1)} jogadores."
            )
        ultimo = min(total - base * (quantidade - 1), base)
        return [base] * (quantidade - 1) + [ultimo]
    base, sobras = divmod(total, quantidade)
    if base < 1:
        raise ValueError("Nao ha jogadores suficientes para essa quantidade de times.")
    return [base + (1 if i < sobras else 0) for i in range(quantidade)]


def _parse_tamanhos(tamanhos):
    if tamanhos is None:
        return None
    if isinstance(tamanhos, str):
        try:
            val = int(tamanhos.strip())
            return [val]
        except ValueError:
            raise ValueError("Digite apenas o tamanho de cada time. Ex.: 8.")
    return list(tamanhos) if tamanhos else None


def _time_nome(index):
    return f"Time {chr(65 + index)}" if index < 26 else f"Time {index + 1}"


def _get(j, attr, default=None):
    val = getattr(j, attr, None)
    if val is not None:
        return val
    if isinstance(j, dict):
        return j.get(attr, default)
    return default


def _tem_posicao(jogador, alvo):
    posicoes = _get(jogador, "posicoes", [])
    if isinstance(posicoes, str):
        return alvo.lower() in posicoes.lower()
    for p in posicoes:
        nome = _get(p, "nome", str(p)).lower()
        if alvo.lower() in nome:
            return True
    return False


def _ids_restritos(jogador):
    restricoes = _get(jogador, "restricoes", [])
    if isinstance(restricoes, (str, int)):
        return []
    ids = set()
    for r in restricoes:
        rid = _get(r, "id", r)
        if rid:
            ids.add(rid)
    return ids


def sortear_times(jogadores, historico=None, tentativas=4000, quantidade=3, tamanhos=None):
    if len(jogadores) < 2:
        raise ValueError("Selecione pelo menos 2 jogadores.")

    # Fase 1: separar goleiros
    goleiros = [j for j in jogadores if _get(j, "is_goleiro", False)]
    jogadores_linha = [j for j in jogadores if not _get(j, "is_goleiro", False)]

    historico = historico or []
    tamanhos_lista = tamanhos_dos_times(len(jogadores), quantidade, _parse_tamanhos(tamanhos))

    if not goleiros:
        # Sem goleiros cadastrados, sorteia tudo junto (improviso)
        goleiros = []
    elif len(goleiros) <= quantidade:
        # Fase 1a: round-robin goleiros para times
        melhor_goleiros_por_time = {}
        # gol is distributed within optimization - skip, handled by _pontuar
        melhor, melhor_pontuacao = None, float("inf")
        melhor_gols = None

        for _ in range(tentativas):
            # Distribute goleiros round-robin
            gols = list(goleiros)
            random.shuffle(gols)
            gols_por_time = {}
            nomes_times = [_time_nome(i) for i in range(quantidade)]
            for i, nome in enumerate(nomes_times):
                gols_por_time[nome] = gols[i] if i < len(gols) else None

            linha = list(jogadores_linha)
            random.shuffle(linha)

            # Build sizes adjusting for goleiros already placed
            times = {nome: [] for nome in nomes_times}
            # Track goleiro IDs to reserve spots
            goleiro_ids_por_time = {}
            for nome, g in gols_por_time.items():
                if g:
                    goleiro_ids_por_time[nome] = g.id

            # Split remaining linha players
            # Remove goleiro count from tamanhos
            tamanhos_linha = []
            for i, nome in enumerate(nomes_times):
                already = 1 if gols_por_time[nome] else 0
                tamanhos_linha.append(tamanhos_lista[i] - already)

            inicio = 0
            for i, nome in enumerate(nomes_times):
                tam = tamanhos_linha[i]
                times[nome] = linha[inicio:inicio + tam]
                inicio += tam

            pontuacao = _pontuar(times, gols_por_time, historico, jogadores)
            if pontuacao < melhor_pontuacao:
                melhor, melhor_pontuacao = times, pontuacao
                melhor_gols = dict(gols_por_time)

        # Montar times finais
        times_finais = {}
        for nome in melhor:
            time = list(melhor[nome])
            if melhor_gols.get(nome):
                time.insert(0, melhor_gols[nome])
            times_finais[nome] = time

        goleiros_finais = melhor_gols
    else:
        # Mais goleiros que times, distribuir e o resto joga na linha
        gols = list(goleiros)
        random.shuffle(gols)
        gols_por_time = {}
        nomes_times = [_time_nome(i) for i in range(quantidade)]
        for i, nome in enumerate(nomes_times):
            gols_por_time[nome] = gols[i]
        goleiros_extras = gols[quantidade:]  # rest as line players
        todos_linha = jogadores_linha + goleiros_extras

        melhor, melhor_pontuacao = None, float("inf")
        melhor_gols = None

        for _ in range(tentativas):
            random.shuffle(todos_linha)
            times = {nome: [] for nome in nomes_times}
            goleiro_ids_time = {}
            for nome, g in gols_por_time.items():
                goleiro_ids_time[nome] = g.id

            tamanhos_linha = [tamanhos_lista[i] - 1 for i in range(quantidade)]
            inicio = 0
            for i, nome in enumerate(nomes_times):
                tam = tamanhos_linha[i]
                times[nome] = todos_linha[inicio:inicio + tam]
                inicio += tam

            pontuacao = _pontuar(times, gols_por_time, historico, jogadores)
            if pontuacao < melhor_pontuacao:
                melhor, melhor_pontuacao = times, pontuacao
                melhor_gols = dict(gols_por_time)

        times_finais = {}
        for nome in melhor:
            time = list(melhor[nome])
            if melhor_gols.get(nome):
                time.insert(0, melhor_gols[nome])
            times_finais[nome] = time

        goleiros_finais = melhor_gols

    # Calculate medias
    medias = {}
    for nome, time in times_finais.items():
        notas = [_get(j, "nota", 0) for j in time]
        medias[nome] = round(sum(notas) / len(notas) if notas else 0, 2)

    return times_finais, goleiros_finais, medias


def _pontuar(times, gols_por_time, historico, todos_jogadores):
    score = 0

    # Build all times with goleiros for scoring
    times_completos = {}
    for nome in times:
        time = list(times[nome])
        g = gols_por_time.get(nome)
        if g:
            time.insert(0, g)
        times_completos[nome] = time

    # Restricao violation = infinite
    restricao_map = {}
    for j in todos_jogadores:
        restricao_map[j.id] = _ids_restritos(j)

    for nome, time in times_completos.items():
        ids = {j.id for j in time}
        for j in time:
            restritos = restricao_map.get(j.id, set())
            if ids & restritos:
                return float("inf")

    # Balanceamento de nota
    totais = []
    for nome, time in times_completos.items():
        totais.append(sum(_get(j, "nota", 0) for j in time))
    if totais:
        media = sum(totais) / len(totais)
        score += sum(abs(t - media) ** 2 for t in totais) * 30

    for nome, time in times_completos.items():
        tem_zagueiro = any(_tem_posicao(j, "Zagueiro") or _tem_posicao(j, "Fixo") for j in time)
        if not tem_zagueiro:
            score += 10000

        atacantes = sum(1 for j in time if _tem_posicao(j, "Atacante") or _tem_posicao(j, "Pivo"))
        score += max(0, atacantes - 3) * 3000

        craques = sum(1 for j in time if _get(j, "nota", 0) == 5)
        score += max(0, craques - 1) * 2500

        iniciantes = sum(1 for j in time if _get(j, "nota", 0) <= 2)
        score += max(0, iniciantes - 1) * 1200

    # Pares repetidos nos ultimos 3 sorteios
    pares = Counter()
    for sorteio in historico[-3:]:
        if isinstance(sorteio, dict):
            for t in sorteio.values():
                for i, j1 in enumerate(t):
                    for j2 in t[i + 1:]:
                        a, b = sorted((str(j1), str(j2)))
                        pares[(a, b)] += 4
        elif isinstance(sorteio, list):
            for t in sorteio:
                ids_t = [str(item) for item in t]
                for i, j1 in enumerate(ids_t):
                    for j2 in ids_t[i + 1:]:
                        a, b = sorted((j1, j2))
                        pares[(a, b)] += 4

    for nome, time in times_completos.items():
        ids = [str(j.id) for j in time]
        for i, j1 in enumerate(ids):
            for j2 in ids[i + 1:]:
                a, b = sorted((j1, j2))
                score += pares.get((a, b), 0)

    return score


def nomes_historico(sorteio):
    grupos = {}
    for item in sorteio.itens:
        grupos.setdefault(item.time, []).append(str(item.jogador_id))
    return list(grupos.values())


def historico_para_engine(sorteios):
    result = []
    for sorteio in sorteios:
```

```python
"""Formatador de resultado para WhatsApp com emojis."""


def _icone(jogador, posicoes_por_jogador):
    pos_nomes = [p.nome.lower() for p in getattr(jogador, "posicoes", [])]
    if getattr(jogador, "is_goleiro", False):
        return "\U0001F9E4"
    if any("zagueiro" in p or "fixo" in p for p in pos_nomes):
        return "\U0001F6E1\uFE0F"
    return "\u26BD"


def _estrelas(jogador):
    nota = getattr(jogador, "nota", 3)
    inteiro = int(nota)
    resto = nota - inteiro
    if resto == 0.5:
        return "\u2605" * inteiro + "\u00BD"
    return "\u2605" * inteiro


def _posicoes_str(jogador):
    posicoes = getattr(jogador, "posicoes", [])
    if not posicoes:
        nomes = [getattr(p, "nome", str(p)) for p in posicoes]
        if nomes:
            return " (" + ", ".join(nomes) + ")"
    return ""


def formatar_whatsapp(times, goleiros, medias):
    """Retorna string formatada para WhatsApp.

    Args:
        times: dict[nome_time, list[Jogador]]
        goleiros: dict[nome_time, int|Jogador|None]
        medias: dict[nome_time, float]

    Returns:
        str com emojis formatado para copiar e colar.
    """
    linhas = ["\u26BD *RACHALAB \u2014 Times da Rodada* \u26BD", ""]

    for nome, time in times.items():
        media = medias.get(nome, 0)
        linhas.append(f"*{nome}* (Media: {media:.1f})")

        for jogador in time:
            icone = _icone(jogador, None)
            est = _estrelas(jogador)
            pos = _posicoes_str(jogador)
            linhas.append(f"{icone} {jogador.nome}{pos} {est}")

        g = goleiros.get(nome)
        if g and hasattr(g, "nome"):
            linhas.append(f"\U0001F9E4 Goleiro: {g.nome}")
        elif g is None:
            linhas.append("\U0001F9E4 Goleiro: Improvisado")

        linhas.append("")

    return "\n".join(linhas)
```

```javascript
export function estrelas(nota) {
  const inteiro = Math.floor(nota);
  const resto = nota - inteiro;
  const stars = '★'.repeat(inteiro);
  return resto === 0.5 ? stars + '½' : stars;
}

export function iconePosicao(jogador) {
  if (jogador.is_goleiro) return '🧤';
  const nomes = (jogador.posicoes || []).map(p => p.nome.toLowerCase());
  if (nomes.some(n => n.includes('zagueiro') || n.includes('fixo'))) return '🛡️';
  return '⚽';
}
```

```javascript
import { qs } from '../utils/dom.js';
import { estrelas } from '../utils/format.js';
import { getHistorico, deleteSorteio } from '../api/sorteios.js';

export async function renderHistoricoPanel(container) {
  container.innerHTML = '<p class="text-muted">Carregando...</p>';

  try {
    const historico = await getHistorico(20);

    if (!historico.length) {
      container.innerHTML = '<p class="empty-state">Nenhum sorteio salvo ainda.</p>';
      return;
    }

    let html = '<section class="panel history-panel">';
    html += '<div class="panel-heading"><h2>Historico</h2></div>';

    for (const s of historico) {
      const data = new Date(s.data).toLocaleString('pt-BR');
      html += `<details class="history-item">`;
      html += `<summary>Sorteio #${s.id} · ${data} <button class="btn-del-sorteio icon-button" data-id="${s.id}" title="Excluir">×</button></summary>`;
      html += '<div class="history-teams">';
      for (const [nome, jogadores] of Object.entries(s.times)) {
        html += `<div><strong>${nome}</strong> (Média ${s.medias[nome]})<small>`;
        html += jogadores.map(j => `${j.nome} ${estrelas(j.nota)}`).join(', ');
        html += '</small></div>';
      }
      html += '</div>';
      html += '<div class="history-goalies">';
      const goleiros = Object.entries(s.goleiros || {});
      if (goleiros.length) {
        html += '<strong>Goleiros</strong><div class="history-goalies-list">';
        html += goleiros.map(([time, nome]) => `🥅 ${time}: ${nome}`).join(', ');
        html += '</div>';
      }
      html += '</div></details>';
    }

    html += '</section>';
    container.innerHTML = html;

    container.querySelectorAll('.btn-del-sorteio').forEach(btn => {
      btn.addEventListener('click', async (e) => {
        e.preventDefault();
        await deleteSorteio(parseInt(btn.dataset.id));
        renderHistoricoPanel(container);
      });
    });

  } catch (err) {
    container.innerHTML = '<p class="text-danger">Erro ao carregar historico.</p>';
  }
}
```
